\frametitle{Graphics APIs}

\begin{block}{}
  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item Graphics APIs are software interfaces that allow programs to communicate with a computer’s GPU
    \item Used for loading textures, applying shaders, and rendering 2D and 3D scenes
    \item Applications do not communicate with the GPU directly
    \item Application $\rightarrow$ Graphics API $\rightarrow$ GPU Driver $\rightarrow$ GPU
  \end{itemize}

\end{block}

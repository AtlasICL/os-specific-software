\frametitle{System calls}

\begin{theorembox}
\textbf{What are system calls?} \\
  \begin{itemize}
    \setlength{\itemsep}{8pt}
    \item The OS provides user programs access to hardware resources through \textbf{system calls}.
    \item These allow programs to create windows, spawn processes, open files, etc...
  \end{itemize}
\end{theorembox}

\end{frame}

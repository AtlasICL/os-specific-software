\frametitle{Assessing Solutions – User experience}
\begin{theorembox}
\textbf{Criteria for assessing solutions}: \\
\begin{itemize}
    \setlength{\itemsep}{8pt}
    \item \textbf{Stability} - Is the program stable? Do crashes occur? 
    \item \textbf{Performance} - Does the system lose any performance compared to bare metal? How much performance is lost?
    \item \textbf{Features} - Are any features missing compared to running to the bare metal? What features?
    \item \textbf{Availability of Updates} - What support can the system expect to have? How do users access updates?
\end{itemize}
\end{theorembox}

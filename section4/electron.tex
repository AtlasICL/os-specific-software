\frametitle{Solutions - Electron}
\begin{redbox}
\textbf{Electron}
\begin{itemize}
    \setlength{\itemsep}{8pt}
    \item Open source framework for building cross-platform desktop applications with a JavaScript runtime \cite{electron-why-page}
    \item 1 language for Windows, Linux and MacOS \cite{electron-why-page}
    \item Translates web interfaces into desktop software rather than translating system calls
\end{itemize}
\end{redbox}
\begin{exbox}
\textbf{Examples}
\begin{itemize}
    \item VS Code, Discord, Slack

\end{itemize}
\end{exbox}

\begin{frame}
\frametitle{Electron - Advantages and Disadvantages}
\begin{theorembox}
\textbf{Advantages}
\begin{itemize}
    \item Cross-platform support for Windows, Linux, and MacOS
    \item Shared web and desktop code $\rightarrow$ code reusability
    \item Open source
\end{itemize}
\end{theorembox}
\begin{redbox}
\textbf{Disadvantages}
\begin{itemize}
    \item High memory usage $\rightarrow$ lower performance compared to native applications
    % say because Chromium and the JavaScript runtime remain active even when idle
    \item Large file sizes $\rightarrow$ Chromium and Node.js are bundled with every application
\end{itemize}
\end{redbox}
\end{frame}

\frametitle{Electron - Evaluation}

\vspace{-30pt}
\begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
        \begin{theorembox}
        \textbf{User experience}
        \vspace{5pt}
        \begin{itemize}
            \setlength{\itemsep}{4pt}
            \item \textbf{Stability}: Can degrade in  unoptimised applications
            \item \textbf{Performance}: Complex layouts and heavy animations increase CPU and GPU usage
            \item \textbf{Features}: Poor support for low-latency tasks (e.g: trading)
            \item \textbf{Updates}: Frequent updates through built-in updaters, often involving large downloads
        \end{itemize}
        \vspace{4pt}
        \end{theorembox}
    \end{column}

    \begin{column}{0.49\textwidth}
        \begin{theorembox}
        \textbf{Developer experience}
        \vspace{5pt}
        \begin{itemize}            
            \setlength{\itemsep}{4pt}
            \item \textbf{Development Time}: Efficient
            \item \textbf{Additional Expertise}: Helpful to understand Electron and web development languages
            \item \textbf{Maintenance}: Moderate
        \end{itemize}
        \vspace{78pt}
        \end{theorembox}
    \end{column}
\end{columns}

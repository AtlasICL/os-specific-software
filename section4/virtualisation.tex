
\begin{frame}
\frametitle{Solutions - Virtualisation}
\begin{redbox}
\textbf{Virtualisation}
\begin{itemize}
    \item Divides physical computer resources into a series of virtual environments. \cite{virtualization-wikipedia}
    \item Software directly shares a host machine's hardware.
    \item This is NOT the same as emulation! \cite{virtualization-wikipedia}
\end{itemize}
\end{redbox}
\begin{yellowbox}
\textbf{Containerisation}
\begin{itemize}
    \item Virtualisation at OS-level. \cite{containers}
    \item Software, along with its prerequisite OS libraries and dependencies, is packaged into a container. \cite{containers}
    \item \textbf{Container:} A single executable that can run consistently on ANY system infrastructure. \cite{containers}
\end{itemize}
\end{yellowbox}

\end{frame}

\begin{frame}
\frametitle{Containerisation - Advantages and Disadvantages}
\begin{theorembox}
\textbf{Advantages}
\begin{itemize}
    \item Improves scalability and utilisation of resources \cite{baresi-containers}
    \item Container overhead $<$ VM overhead \cite{baresi-containers}
    \item Host's kernel shared by all containers \cite{kernels}
    \item Each container isolated during its runtime \cite{kernels}
\end{itemize}
\end{theorembox}
\begin{redbox}
\textbf{Disadvantages}
\begin{itemize}
    \item Kernel vulnerability $\rightarrow$ security risk to all containers \cite{kernel-risk}
    \item Complex to manage large volumes of containers
    \item Suboptimal performance for resource-intensive workloads \cite{baresi-containers}
\end{itemize}
\end{redbox}
\end{frame}

        \vspace{5pt}
        \begin{itemize}
            \item \textbf{Stability} - Very stable
            \item \textbf{Performance} - Excellent due to fewer overheads
            \item \textbf{Features} - Highly portable

            \item \textbf{Additional Expertise} - Helpful to understand container tools
